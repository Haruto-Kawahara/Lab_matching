js:
const questions = {
 start:{text:"",yes:"q1",no:"q2"},
 q1:{text:"",yes:"lab1",no:"lab2"},
 q2:{text:"",yes:"q3",no:"q4"},
 q3:{text:"",yes:"lab3",no:"lab4"},
 q4:{text:"",yes:"lab5",no:"lab6"},
}
 const labs ={
  lab1:{name:"",description:"",reason:""},
  lab2:{name:"",description:"",reason:""},
  lab3:{name:"",description:"",reason:""},
  lab4:{name:"",description:"",reason:""},
  lab5:{name:"",description:"",reason:""},
  lab6:{name:"",description:"",reason:""},
}
let currentQuestion = "start";
function answer(choice) {
    currentQuestion = questions[currentQuestion][choice];
    showQuestion();
}
function showQuestion() {
    if (labs[currentQuestion]) {
        const lab = labs[currentQuestion];
        document.getElementById("question").textContent =
            lab.name + "がおすすめです！";
        document.getElementById("description").textContent =
            lab.description;
        document.getElementById("reason").textContent =
            lab.reason;
        // YES・NOボタンを隠す
        document.getElementById("buttons").style.display = "none";
        // 診断結果を表示する
        document.getElementById("result").style.display = "block";
        return;
    }
    const question = questions[currentQuestion];
    document.getElementById("question").textContent =
        question.text;
    document.getElementById("buttons").style.display = "block";
    document.getElementById("result").style.display = "none";
}
//最初の質問を表示する
showQuestion();
